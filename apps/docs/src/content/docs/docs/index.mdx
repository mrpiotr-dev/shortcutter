---
title: Getting started
description: Install shortcutter and register your first keyboard shortcut.
---

```sh
pnpm add @shortcutter/core
```

```ts
import { createShortcutter } from '@shortcutter/core';

const shortcutter = createShortcutter();

shortcutter.listen('ControlOrMeta+K', () => openCommandPalette(), { preventDefault: true });
```
